// resolve-hook.mjs
import { pathToFileURL } from "node:url";
const STUB_DB = process.env.T7F1_STUB_DB;
export async function resolve(specifier, context, nextResolve) {
  if (STUB_DB && /(^|\/)shared\/db$/.test(specifier)) {
    return { url: pathToFileURL(STUB_DB).href, shortCircuit: true };
  }
  try { return await nextResolve(specifier, context); }
  catch (err) {
    if (specifier.startsWith(".") && !/\.(m?js|cjs|ts|json)$/.test(specifier)) {
      for (const suffix of [".ts", "/index.ts"]) {
        try { return await nextResolve(specifier + suffix, context); } catch {}
      }
    }
    throw err;
  }
}
// register.mjs
import { register } from "node:module";
register("./resolve-hook.mjs", new URL("./", import.meta.url));   // 第二参数必须是目录 URL
