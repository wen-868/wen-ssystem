> 用途：只在"node 不能起子进程"的沙箱里执行**真实门禁文件**（Node 原生 TS type-stripping + 极简 `expect`），不修改门禁文件本身。

`_verify-119f1-vitest-shim.mjs`

```js
export const stats = { passed: 0, failed: 0 };
export function describe(name, fn) { console.log(`\n> ${name}`); fn(); }
export function it(name, fn) {
  try { fn(); stats.passed++; console.log(`  ok   ${name}`); }
  catch (err) { stats.failed++; console.log(`  FAIL ${name}`); console.log(`       ${String(err && err.message).split("\n").join("\n       ")}`); }
}
export function expect(actual, message) {
  const fail = (detail) => { const err = new Error(`${message ? message + "\n" : ""}${detail}`); err.name = "AssertionError"; throw err; };
  return {
    toBeGreaterThanOrEqual(n) { if (!(actual >= n)) fail(`expected ${actual} to be >= ${n}`); },
    toEqual(other) { const a = JSON.stringify(actual), b = JSON.stringify(other); if (a !== b) fail(`expected ${a}\n  to equal ${b}`); },
  };
}
```
