### F1-三、前缀归一的处理与理由

1. **两种字面写法并存，必须归一后才能逐字比较**：`saas-admin/src/api.ts` 里写的是 `/platform/reports/export…`（该实例 `baseURL=/api`，运行期真实请求路径 = `/api` + 该字面），`Dashboard.vue` 的契约注释写的是完整 `/api/platform/reports/export…`。
2. **处理**：正则用可选前缀组 `(?:\/api)?\/platform\/reports\/export…` 同时吃掉两种写法，再对每条命中做两次归一 —— `/\$\{[^}]*\}/ → ":id"`（模板串参数）与 `^\/platform\/ → "/api/platform/"`（补 `baseURL` 前缀）；最后 `Set` 去重 + `sort()`。
3. **为什么不用"两种写法都算不同路径"或"只放宽成包含匹配"**：前者会导致同一端点被算成 2 条（`4 + 4 = 8`），断言必红且判据失真；后者（扫全仓 / 命中即算 / `includes` 松匹配）会把"前端到底用了哪几条"这个判据彻底抹掉 —— 正是卡内红线①禁止的"为变绿而放宽"。归一后 `expect(frontend).toEqual([4 条])` 的**逐字**强度与 F1 之前完全一致。
4. **补一条硬约束（防"文件再搬一次就静默变弱"）**：`locateExportContractSources()` 对清单内**每个**文件都要求存在，缺一个**抛错**（而不是"扫到几算几"）。若将来 4 条字面路径再搬家（例如再抽一层 `api/platform/*.ts`），契约用例会以明确报错判红，逼后来人同步清单 —— 不允许静默降级。
5. **不扫全仓的理由（为何这不算"松口径"）**：字面路径的**真实使用点**就这两个文件（`rg -n "platform/reports/export" saas-admin/src` 只命中 `api.ts` 与 `Dashboard.vue`）；白名单式双文件比全仓扫描更严（全仓扫描会把文档/测试/注释里的任何字符串都算成"前端在用"）。

```js
import fs from 'node:fs'
import os from 'node:os'
import path from 'node:path'
import { pathToFileURL } from 'node:url'

const ROOT = path.resolve(process.argv[2] ?? '.')
const TEST_FILE = path.join(ROOT, 'backend/src/__tests__/routes/platform-export-task.test.ts')
const ROUTES_FILE = path.join(ROOT, 'backend/src/routes/platform-export-task.routes.ts')
const API_TS = path.join(ROOT, 'saas-admin/src/api.ts')
const DASH_VUE = path.join(ROOT, 'saas-admin/src/views/Dashboard.vue')

/** 契约用例 `expect(frontend).toEqual([...])` 的期望值，逐字照抄 */
const EXPECTED_FRONTEND = [
  '/api/platform/reports/export',
  '/api/platform/reports/export/:id/download',
  '/api/platform/reports/export/:id/logs',
  '/api/platform/reports/export/:id/status',
]
const EXPECTED_BACKEND_ONLY = ['/api/platform/reports/export/:id/retry']

/* ── 1) 切出真实提取器源码 → 临时 .ts 模块 → import ── */
const START = '/* >>>F1-CONTRACT-EXTRACTOR-START<<< */'
const END = '/* >>>F1-CONTRACT-EXTRACTOR-END<<< */'

async function loadExtractor() {
  const src = fs.readFileSync(TEST_FILE, 'utf8')
  const a = src.indexOf(START)
  const b = src.indexOf(END)
  if (a < 0 || b < 0) throw new Error(`测试文件缺少提取器切片标记：${TEST_FILE}`)
  const body = src.slice(a + START.length, b)
  const gen = fs.mkdtempSync(path.join(os.tmpdir(), 'c630-f1-gen-'))
  const file = path.join(gen, 'extractor.ts')
  fs.writeFileSync(
    file,
    [
      'import { existsSync, readFileSync } from "node:fs";',
      'import { resolve } from "node:path";',
      body,
      'export { frontendExportContract, locateExportContractSources };',
    ].join('\n'),
    'utf8'
  )
  console.log(`[装置] 提取器原文切片：${TEST_FILE}`)
  console.log(`[装置] 生成临时模块：${file}（${body.length} 字符）`)
  return import(pathToFileURL(file).href)
}

/* ── 2) 后端注册全量路径（静态解析真实路由文件，口径 = 测试里 declaredFullPaths()） ── */
function backendFullPaths() {
  const src = fs.readFileSync(ROUTES_FILE, 'utf8')
  const prefixes = [...src.matchAll(/prefix:\s*"([^"]+)"/g)].map((m) => m[1])
  const rels = [...src.matchAll(/platformExportTaskRouter\.(get|post|put|delete|patch)\(\s*"([^"]+)"/g)].map(
    (m) => m[2]
  )
  const full = new Set()
  for (const prefix of prefixes) {
    for (const rel of rels) full.add(prefix + (rel === '/' ? '' : rel))
  }
  return [...full].sort()
}

const sameList = (a, b) => JSON.stringify(a) === JSON.stringify(b)

/* ── 3) 复刻契约用例的两条断言 ── */
function contractAssertions(frontend) {
  const backend = backendFullPaths()
  const a1 = sameList(frontend, EXPECTED_FRONTEND)
  const backendOnly = backend.filter((p) => !new Set(frontend).has(p))
  const a2 = sameList(backendOnly, EXPECTED_BACKEND_ONLY)
  return { a1, a2, backend, backendOnly }
}

function run(label) {
  const frontend = extractor.frontendExportContract()
  const r = contractAssertions(frontend)
  const red = !(r.a1 && r.a2)
  console.log(`\n=== ${label} ===`)
  console.log(`cwd = ${process.cwd()}`)
  console.log(`定位到的契约文件 = ${JSON.stringify(extractor.locateExportContractSources())}`)
  console.log(`frontendExportContract() = ${JSON.stringify(frontend)}`)
  if (r.a1) {
    console.log('✓ 断言1 绿：前端 4 条字面路径恰好等于契约')
  } else {
    console.log(
      `✗ 断言1 红：AssertionError: expected ${JSON.stringify(frontend)} to deeply equal ${JSON.stringify(EXPECTED_FRONTEND)}`
    )
  }
  if (r.a2) {
    console.log('✓ 断言2 绿：后端全量 = 前端 4 条 + /:id/retry（无第 7 条）')
  } else {
    console.log(
      `✗ 断言2 红：AssertionError: expected ${JSON.stringify(r.backendOnly)} to deeply equal ${JSON.stringify(EXPECTED_BACKEND_ONLY)}`
    )
  }
  console.log(`  后端全量注册路径 = ${JSON.stringify(r.backend)}`)
  console.log(`结论：${red ? 'RED（门禁能红）' : 'GREEN（门禁回绿）'}`)
  return red
}

/* ── 4) 造错沙箱：只把 api.ts 的 ${id}/status 改成 ${id}/statuss ── */
const WORK = fs.mkdtempSync(path.join(os.tmpdir(), 'c630-f1-sbx-'))
function buildBrokenSandbox() {
  fs.mkdirSync(path.join(WORK, 'saas-admin/src/views'), { recursive: true })
  const apiSrc = fs.readFileSync(API_TS, 'utf8')
  const needle = '/platform/reports/export/${id}/status'
  const broken = apiSrc.replace(needle, '/platform/reports/export/${id}/statuss')
  if (broken === apiSrc) throw new Error(`造错失败：api.ts 未匹配到 ${needle}`)
  fs.writeFileSync(path.join(WORK, 'saas-admin/src/api.ts'), broken, 'utf8')
  fs.copyFileSync(DASH_VUE, path.join(WORK, 'saas-admin/src/views/Dashboard.vue'))
  console.log(`\n[装置] 造错沙箱 = ${WORK}`)
  console.log('[装置] 沙箱 api.ts 相对真件只改了 1 条字面路径：/platform/reports/export/${id}/status → /statuss')
}

const extractor = await loadExtractor()
buildBrokenSandbox()

process.chdir(WORK)
const redWhenBroken = run('反测 ①｜api.ts /status → /statuss（应当红）')

process.chdir(ROOT)
const redWhenRestored = run('反测 ②｜复原（真实仓库原文件，应当绿）')

const pass = redWhenBroken && !redWhenRestored
console.log(
  `\n总结：改坏 ⇒ ${redWhenBroken ? '红 ✓' : '绿 ✗（门禁是假的）'}；复原 ⇒ ${redWhenRestored ? '红 ✗' : '绿 ✓'}`
)
console.log(`总判定：${pass ? 'PASS（该门禁能红、且复原能绿）' : 'FAIL'}`)
process.exitCode = pass ? 0 : 1
```

1. **自检装置未入库**（运行于工作树根临时文件、已删除）：因为卡内 F1-④要求 `git diff --numstat` **只出现 3 个文件**，与"新增一个 `saas-admin/tools/*.mjs`"（F1-②括号里的"参照 `c6-3-0-reverse-test.mjs` 形态"）冲突。我选了**遵 numstat 硬约束**、把装置全文附在 F1-四 里保证可复跑。**若凌舟要求它入库**，请回一句，我把同一份文件落到 `saas-admin/tools/c6-3-0-f1-extractor-selfcheck.mjs`（届时 numstat 会多 1 个新增文件）。

4. **`declaredFullPaths()` 的运行期读数未在本机复现**：自检装置用静态解析近似（本工作树无法 import express），故 F1-四 的两条断言是"同口径复刻"；CI 里跑的是真运行期读数。
