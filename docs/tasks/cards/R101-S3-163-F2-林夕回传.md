/** 纯字符串扫描取出 .vue 的 script 段：只取第一个 <script…> 与 </script…> 之间的内容（找不到返回空串）。
 *  全程用 indexOf/slice 定位，不使用任何正则，避免再次被 CodeQL js/bad-html-filtering-regexp 启发式命中。 */
function scriptSectionOf(raw) {
  const lower = raw.toLowerCase()
  const open = lower.indexOf('<script')
  if (open === -1) return ''
  const gt = raw.indexOf('>', open)
  if (gt === -1) return ''
  const close = lower.indexOf('</script', gt)
  if (close === -1) return ''
  return raw.slice(gt + 1, close)
}

/** .vue 只取 script 段（模板里的同名字符串不算实现） */
function codeOf(file, raw) {
  if (file.endsWith('.vue')) return scriptSectionOf(raw)
  return raw
}

## 三、为什么这样改就不会再触发那条 CodeQL 规则

`js/bad-html-filtering-regexp` 的候选对象**只有正则字面量/正则对象**——它逐条检查正则里对 HTML 标签的匹配是否"覆盖了全部形态（大小写、属性、空白、自闭合…）"，所以只要用正则去匹配 `<script…>` / `</script…>`，它就会不断找出下一个"未覆盖形态"（本行已连续两轮分别报"未覆盖大写 `<SCRIPT>`"和"未覆盖 `</script >` 这种带空格收尾"）。

改后提取流程里**一次正则都没有**：定位用 `String.prototype.indexOf`（对小写化副本做大小写无关定位），取值用 `String.prototype.slice`。文件里其余正则（第 45/92/98/100/110/113/115/119/120/132/133 行）均与 HTML/script 标签无关（注释剥离、函数名匹配、清 token 动作匹配），因此该规则**没有候选可分析**，从根上不再命中，且**未使用任何抑制手段**。
