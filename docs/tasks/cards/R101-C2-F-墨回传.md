import { createRequire } from 'node:module'

const require = createRequire('D:/Users/ZXQL/wt-agents/issue-61/package.json')

  }

function loadModule(file) {
  const out = ts.transpileModule(fs.readFileSync(file, 'utf8'), {
    compilerOptions: { module: ts.ModuleKind.CommonJS, target: ts.ScriptTarget.ES2020, esModuleInterop: true },
    fileName: file,
  }).outputText
  const module = { exports: {} }

  }
  new Function('exports', 'require', 'module', '__filename', '__dirname', out)(module.exports, fakeRequire, module, file, path.dirname(file))
  return module.exports
}

const tpl = loadModule(path.join(API_DIR, 'platform-template.ts'))
const ann = loadModule(path.join(API_DIR, 'announcement.ts'))
