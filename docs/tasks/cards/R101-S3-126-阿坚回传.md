export async function resolve(specifier, context, next) {
  let url;
  try {
    url = (await next(specifier, context)).url;
  } catch (err) {
    if (!(specifier.startsWith("./") || specifier.startsWith("../")) || !context.parentURL?.startsWith("file:")) throw err;
    const base = path.dirname(fileURLToPath(context.parentURL));
    let found = null;
    for (const suffix of [".ts", "/index.ts", ".js", ".mjs"]) {
      const cand = path.resolve(base, specifier + suffix);
      try { if ((await stat(cand)).isFile()) { found = cand; break; } } catch {}
    }
    if (!found) throw err;
    url = pathToFileURL(found).href;
  }

  const parent = context.parentURL || "";
  const isRbacStub = parent.endsWith("rbac-stub.mjs");

  if (!isRbacStub && url.endsWith("/src/services/admin/rbac.service.ts")) {
    return { url: pathToFileURL(T + "rbac-stub.mjs").href, shortCircuit: true };
  }
  if (!url.startsWith("file:")) return { url, shortCircuit: true };
  for (const [pat, target] of REDIRECTS) {
    if (url.endsWith(pat)) return { url: pathToFileURL(target).href, shortCircuit: true };
  }
  return { url, shortCircuit: true };
}

// ---- load 钩子：仅对 __tests__/mocks/**.ts 做「类型名已被 Node 类型擦除但 import 仍在」的收缩 ----
export async function load(url, context, next) {
  if (url.includes("/__tests__/mocks/") && url.endsWith(".ts")) {
    const file = fileURLToPath(url);
    let src = readFileSync(file, "utf8");
    const dir = path.dirname(file);
    src = src.replace(/import\s*\{([^}]*)\}\s*from\s*"(\.[^"]*)";/g, (full, names, spec) => {
      const base = path.resolve(dir, spec);
      let target = null;
      for (const s of [".ts", "/index.ts", ""]) if (existsSync(base + s)) { target = base + s; break; }
      if (!target) return full;
      const tsrc = readFileSync(target, "utf8");
      const runtime = new Set();
      for (const m of tsrc.matchAll(/export\s+(?:async\s+)?(?:const|let|var|function|class|enum)\s+([A-Za-z_$][\w$]*)/g)) runtime.add(m[1]);
      for (const m of tsrc.matchAll(/export\s*\{([^}]*)\}/g)) for (const n of m[1].split(",")) { const t = n.trim().split(/\s+as\s+/).pop().trim(); if (t) runtime.add(t); }
      const list = names.split(",").map((n) => n.trim()).filter(Boolean);
      const kept = list.filter((n) => {
        if (n.startsWith("type ")) return true;
        const id = n.split(/\s+as\s+/)[0].trim();
        return runtime.has(id);
      });
      if (kept.length === list.length) return full;
      return `import { ${kept.join(", ")} } from "${spec}";`;
    });
    return { format: "module-typescript", source: src, shortCircuit: true };
  }
  return next(url, context);
}
