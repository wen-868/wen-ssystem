## 1. S3-66 方法（可复跑口径）

- **抽取**：解析 `admin-web/src/**/*.{vue,css}`（178 文件）内全部 `<style>` 块与 .css；共 2584 条规则、7833 条声明，取 `prop === "color"` 的声明 **749** 条，其中前景可解析 **745** 条。
- **token 回查**：`var(--x)` 递归解析到 `tokens.css` 内定义值（多层 var 链逐层展开）；无法解析的标"前景未定性"（4 条）。
- **中性/彩色划分**：以 `max(R,G,B) − min(R,G,B) ≤ 16` 判中性（灰阶，天然高对比、非本卡对象），剔除 **509** 条；**彩色前景 = 236 条**（本卡口径对象）。
- **surface（底色）**：只用源码证据——① 同规则自声明背景；② **模板祖先链**（解析 `<template>` 的标签/class/`:class` 绑定，逐级向上找不透明背景规则，含半透明层按 alpha 合成）；③ 走到根仍无 → 记 `body` 兜底来源（`styles.css:56 body{background:var(--bg-page)}`）。
  - 若祖先链上出现 **EP 组件**（`el-card`/`el-dialog`/`el-form-item`/`el-descriptions-item`/`el-table-column` 等）而该项目源码未声明其内部背景 → 判 **surface 未定性**（EP 内部样式不在本次允许读取范围，**不猜**）。
  - 若选择器未匹配到任何模板元素（EP 内部类 / `:deep()` 目标类不在模板里）→ 判 **surface 未知**，单列。
- **字号/字重**：同文件（scoped）+ 全局样式内，按"class 集合子集"匹配同一元素的 font-size/font-weight（支持 BEM 修饰类与父类分行写法）；取不到时 **从严按普通文本（4.5）**，并在表中标注"未定性"。
- **阈值**：普通 ≥4.5:1；大文本（≥24px，或 ≥18.66px 且粗体）≥3:1（WCAG 1.4.3）。
- **前序口径复现**：族名过滤 `metric|kpi|stat|kv|num|value|amount|count|total|summary|figure`（取选择器**末段**）+ 对 #FFFFFF/#F0F0F0/#FAFAFA/#F5F5F5 取**最严**比值。

**依据与差异归因（可核对）**：
1. **前序扫描脚本未入库**：`git log --all --diff-filter=A --name-only -- ".*contrast*"` 仅见 `admin-web/scripts/contrast-metric.cjs`（单元素复算脚本，S3-64 提交 578234d4/6d23380b），**没有全量扫描脚本**，因此无法逐条对齐口径。
2. **计数单位可能不同**：本卡按"规则（声明）条数"计数，同选择器出现一次计 1；若前序按"元素×规则"计数或多计了 `.el-*` 全局规则，会抬高总数（本卡全局样式规则 26 条计入总数）。
3. **中性判定阈值不同**：本卡用 `max−min ≤ 16` 剔灰阶；若前序用"白名单式中性表"，靠近中性的彩色（如 `#596/…`）归属会漂移。
4. **达标/不达标的阈值差异**：本卡对"字号未定性"从严按 4.5（与前序"从严待复测"口径一致）；但本卡从**同文件/全局样式**补齐了大量字号（BEM 修饰类 + 父类分行），部分前序判 FAIL 的条目在补到 ≥24px/粗体后按大文本 3:1 转 PASS（这解释了"达标 +2 / 不达标 −17"的方向）。
5. **方法学不一致（关键）**：前序的 119/51/68 建立在"候选底取最严"的**假设**上；本卡按派单要求"surface 取不到不许猜"，因此**不再给出一个替代的 119**，而是给出"可证 111 + 未定性 53 + 未知 60 + 多义 12"的分层结论。

const fs = require('fs'), path = require('path'), os = require('os');

function walk(dir, out = []) { for (const e of fs.readdirSync(dir, { withFileTypes: true })) { const p = path.join(dir, e.name); if (e.isDirectory()) walk(p, out); else if (/\.(vue|css)$/.test(e.name)) out.push(p); } return out; }

const countNl = (s) => { let n = 0; for (let i = 0; i < s.length; i++) if (s[i] === '\n') n++; return n; };

function parseCss(text, file, startLine) {
  const rules = [], decls = [];
  let i = 0, line = startLine; const n = text.length; const stack = []; let prelude = '', preludeLine = startLine;
  const curSel = () => { for (let k = stack.length - 1; k >= 0; k--) if (stack[k].kind === 'sel') return stack[k].full; return null; };
  const inKf = () => stack.some((f) => f.kind === 'at' && /@keyframes/i.test(f.sel));
  const pushDecl = (raw, atLine) => { const idx = raw.indexOf(':'); if (idx < 0) return; const prop = raw.slice(0, idx).trim().toLowerCase(); const value = raw.slice(idx + 1).trim(); if (!prop) return; decls.push({ file, line: atLine, prop, value, sel: curSel(), inKeyframes: inKf() }); };
  while (i < n) {
    const ch = text[i];
    if (ch === '/' && text[i + 1] === '*') { let j = text.indexOf('*/', i + 2); j = j < 0 ? n : j + 2; line += countNl(text.slice(i, j)); i = j; continue; }
    if (ch === '"' || ch === "'") { const q = ch; let j = i + 1; while (j < n) { if (text[j] === '\\') { j += 2; continue; } if (text[j] === '\n') line++; if (text[j] === q) break; j++; } prelude += text.slice(i, j + 1); i = j + 1; continue; }
    if (ch === ';') { pushDecl(prelude, preludeLine); prelude = ''; preludeLine = line; i++; continue; }
    if (ch === '{') {
      const sel = prelude.trim().replace(/\s+/g, ' ');
      if (sel.startsWith('@')) stack.push({ kind: 'at', sel, full: null });
      else { const parent = curSel(); const full = parent ? parent + ' ' + sel : sel; const rec = { file, line: preludeLine, selectorText: sel, full, atStack: stack.filter((f) => f.kind === 'at').map((f) => f.sel), inKeyframes: inKf() }; stack.push({ kind: 'sel', sel, full, rec }); rules.push(rec); }
      prelude = ''; preludeLine = line; i++; continue;
    }
    if (ch === '}') { pushDecl(prelude, preludeLine); stack.pop(); prelude = ''; preludeLine = line; i++; continue; }
    if (ch === '\n') { line++; if (!prelude.trim()) preludeLine = line; }
    prelude += ch; i++;
  }
  return { rules, decls };
}
function cssSegments(file, text) {

function parseColor(v, depth = 0) {

  m = s.match(/^rgba?\(([^)]+)\)$/i);
  if (m) { const parts = m[1].split(/[\s,\/]+/).filter(Boolean); const num = (x, sc = 255) => x.endsWith('%') ? function parseTemplate(file, text) {
  const els = [];

  let i = tm.index + tm[0].length;
  const endIdx = text.lastIndexOf('</template>'); const stop = endIdx > i ? endIdx : text.length;
  const stack = [];
  while (i < stop) {
    const lt = text.indexOf('<', i); if (lt < 0 || lt >= stop) break;
    if (text.startsWith('<!--', lt)) { const e = text.indexOf('-->', lt); i = e < 0 ? stop : e + 3; continue; }
    const m = /^<(\/?)([A-Za-z][\w.:-]*)/.exec(text.slice(lt, lt + 60)); if (!m) { i = lt + 1; continue; }
    const closing = m[1] === '/', tag = m[2];
    let j = lt + m[0].length, attrStr = '', selfClose = false;
    while (j < stop) { const c = text[j]; if (c === '"' || c === "'") { const q = c; let k = j + 1; while (k < stop && text[k] !== q) k++; attrStr += text.slice(j, k + 1); j = k + 1; continue; } if (c === '>') { selfClose = text[j - 1] === '/'; break; } attrStr += c; j++; }
    const line = countNl(text.slice(0, lt)) + 1;
    if (closing) { for (let k = stack.length - 1; k >= 0; k--) if (stack[k].tag === tag) { stack.length = k; break; } i = j + 1; continue; }
    const classes = new Set(); let dynamic = false;
    const classAttr = attrStr.match(/(?:^|\s)class\s*=\s*("([^"]*)"|'([^']*)')/);
    if (classAttr) (classAttr[2] || classAttr[3] || '').split(/\s+/).filter(Boolean).forEach((c) => classes.add(c));
    const bindAttr = attrStr.match(/(?:^|\s)(?::class|v-bind:class)\s*=\s*("([^"]*)"|'([^']*)')/);
    if (bindAttr) { const val = bindAttr[2] || bindAttr[3] || ''; for (const t of (val.match(/['"]([^'"]+)['"]/g) || [])) { const c = t.slice(1, -1).trim(); if (/^[A-Za-z][\w-]*$/.test(c)) classes.add(c); } if (/\$\{/.test(val) || /[A-Za-z_$][\w$]*\s*[?.[]/.test(val)) dynamic = true; }

  }

}

/* ---- v3 修正：font 解析（含 px/rem/bold/大文本判定 + 同文件/全局作用域过滤） ---- */
function fontOf(ruleSel, classSets, declFile) {
  const res = { size: null, weight: null, sizeFrom: null, weightFrom: null };
  const hits = [];
  for (const [sel, ds] of bySel) {
    const c = compoundClasses(sel);
    if (!c.classes.length) continue;
    if (!c.classes.every((x) => classSets.some((S) => S.has(x)))) continue;
    for (const d of ds) {
      if (!isGlobalFile(d.file) && d.file !== declFile) continue;
      hits.push({ sel, d, specificity: c.classes.length, line: d.line });
    }
  }
  hits.sort((a, b) => (a.specificity - b.specificity) || (a.line - b.line));
  for (const h of hits) {
    if (h.d.prop === 'font-size' && !res.size) { res.size = h.d.value; res.sizeFrom = h.d.file + ':' + h.d.line + ' ' + h.sel; }
    if (h.d.prop === 'font-weight' && !res.weight) { res.weight = h.d.value; res.weightFrom = h.d.file + ':' + h.d.line + ' ' + h.sel; }
  }
  const pxM = res.size && String(res.size).match(/([\d.]+)px/);
  const remM = res.size && String(res.size).match(/([\d.]+)rem/);
  res.fs = pxM ? parseFloat(pxM[1]) : (remM ? parseFloat(remM[1]) * 16 : null);
  res.bold = res.weight ? (parseInt(res.weight, 10) >= 600 || /bold/i.test(res.weight)) : false;
  res.large = res.fs != null && (res.fs >= 24 || (res.fs >= 18.66 && res.bold));
  return res;
}
