## 附录 E：本地验证脚手架（可复跑，不属于仓库文件）

> 本沙箱无 bash/WSL、无 MySQL、且 `child_process.spawn` 被沙箱拒（EPERM），因此脚手架把"桩 API + 逐个场景加载冒烟脚本"放在**同一 Node 进程**内：
> 桩记录每个请求的方法+路径，用动态 `import(...?scenario=N)` 规避 ESM 缓存，用 `process.exit` 钩子取真实退出码。
> 复跑：把下列内容存为任意临时文件（本次在 `C:\Users\XIONG\AppData\Local\Temp\s3-118\stub-readonly-check.mjs`），执行
> `node <文件> <仓库根目录>`，期望：场景1 `[非白名单写请求] 0 条 ✅`／场景2 `[退出码] 1`／场景3 `[退出码] 1` 且桩只收到 `GET /health`。

```js
import http from "node:http";
import path from "node:path";
import { pathToFileURL } from "node:url";

const REPO = process.argv[2];
if (!REPO) { console.error("用法：node stub-readonly-check.mjs <仓库根目录>"); process.exit(2); }
const SMOKE = path.join(REPO, "scripts", "mysql-smoke-test.mjs");
const PORT = Number(process.env.STUB_PORT || 18099);
const ACCEPTED_PASSWORD = "STUB_ACCEPTED_PASSWORD";
let requests = [];

const server = http.createServer((req, res) => {
  let body = "";
  req.on("data", (c) => { body += c; });
  req.on("end", () => {
    requests.push(req.method + " " + req.url);
    const send = (status, payload) => { res.writeHead(status, { "content-type": "application/json" }); res.end(JSON.stringify(payload)); };
    if (req.method === "GET" && req.url === "/health") return send(200, { code: "0", data: { service: "s3-118-stub" } });
    if (req.method === "POST" && req.url === "/api/admin/auth/login") {
      let password = "";
      try { password = JSON.parse(body).password; } catch { password = ""; }
      if (password !== ACCEPTED_PASSWORD) return send(401, { code: "401", msg: "用户名或密码错误" });
      return send(200, { code: "0", data: { token: "stub-token", csrfToken: "stub-csrf" } });
    }
    if (req.method !== "GET") return send(500, { code: "500", msg: "桩服务端：只读档不应发出写请求" });
    if (!req.headers.authorization) return send(401, { code: "401", msg: "未登录" });
    if (req.url.startsWith("/api/admin/products") || req.url.startsWith("/api/miniapp/products")) return send(200, { code: "0", data: { records: [] } });
    return send(200, { code: "0", data: { ok: true } });
  });
});

const realLog = console.log;
const realError = console.error;
const realExit = process.exit;

async function runScenario(label, env, index) {
  requests = [];
  const lines = [];
  console.log = (...a) => lines.push(a.join(" "));
  console.error = (...a) => lines.push("STDERR " + a.join(" "));
  let exitCode = 0;
  let thrown = null;
  process.exit = (code) => { const e = new Error("EXIT"); e.exitCode = code; throw e; };
  process.argv = [process.execPath, SMOKE, "--readonly"];
  for (const key of ["SMOKE_ADMIN_USERNAME", "SMOKE_ADMIN_PASSWORD"]) { delete process.env[key]; }
  Object.assign(process.env, { API_BASE: "http://127.0.0.1:" + PORT, DB_HOST: "127.0.0.1", DB_PORT: "3306" }, env);
  try {
    await import(pathToFileURL(SMOKE).href + "?scenario=" + index);
  } catch (error) {
    if (error.exitCode !== undefined) { exitCode = error.exitCode; } else { thrown = error; }
  } finally {
    console.log = realLog; console.error = realError; process.exit = realExit;
  }
  realLog("\n==================== " + label + " ====================");
  realLog(lines.join("\n"));
  if (thrown) realLog("[异常] " + (thrown && thrown.stack));
  realLog("[退出码] " + exitCode);
  realLog("[桩收到的请求] " + JSON.stringify(requests));
  const writes = requests.filter((r) => !r.startsWith("GET ") && !r.startsWith("POST /api/admin/auth/login"));
  realLog("[非白名单写请求] " + (writes.length === 0 ? "0 条 ✅" : JSON.stringify(writes) + " 🔴"));
  return exitCode;
}

await new Promise((r) => server.listen(PORT, "127.0.0.1", r));
realLog("桩 API 已监听 http://127.0.0.1:" + PORT + "（记录每个请求，非 GET 一律 500）");
await runScenario("场景 1：口令正确（只读档，应仅有登录 POST 一条写语义请求）", { SMOKE_ADMIN_USERNAME: "smoke_bot", SMOKE_ADMIN_PASSWORD: ACCEPTED_PASSWORD }, 1);
await runScenario("场景 2：口令错误（门禁反测，必须非零退出）", { SMOKE_ADMIN_USERNAME: "smoke_bot", SMOKE_ADMIN_PASSWORD: "WRONG_PASSWORD" }, 2);
await runScenario("场景 3：未配置 SMOKE_ADMIN_*（必须明确失败，且不发登录请求）", { SMOKE_ADMIN_USERNAME: "", SMOKE_ADMIN_PASSWORD: "" }, 3);
server.close();
```

> 复跑注意：写入档对照（附录 B.3）用 `STUB_PORT=<桩端口> SMOKE_ALLOW_WRITE=1 SMOKE_ADMIN_USERNAME=smoke_bot SMOKE_ADMIN_PASSWORD=STUB_ACCEPTED_PASSWORD npm run test:mysql:write`，
> 桩会对 3 个 POST 回 500（❌），这正说明写请求只在写入档发出。
