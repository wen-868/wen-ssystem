[31m+ /**[39m
[31m+  * 后端商品列表行 → `ProductInfo`（**SKU 粒度：一条 SKU 一行**）[39m
[31m+  *[39m
[31m+  * 依据（只读后端）：`backend/src/services/admin/product.service.ts:258-301`（有关键词）[39m
[31m+  * 与 `:307-339`（无关键词）两条 SQL 都是 `FROM t_product_sku s JOIN t_product_spu p ...`，[39m
[31m+  * select 同时给出 `p.id AS spuId` 与 `s.id AS skuId`，`total` 也按 `COUNT(*) FROM t_product_sku` 算[39m
[31m+  * ⇒ **列表本身就不是 SPU 维度**，前端按 spuId 合并会造出"一行代表多个规格"的假行。[39m
[31m+  *[39m
[31m+  * 处置（QODER-26）：[39m
[31m+  * 1. `id`/`skuId` 取 `r.skuId`（SKU 行主键），`spuId` 取 `r.spuId`，三者语义分离；[39m
[31m+  * 2. 不再按 spuId 合并、不再跨 SKU 累加库存（原 :319-328 的 merged 逻辑整段废除）；[39m
[31m+  * 3. 缺正整数 skuId 的行**跳过并写守卫日志**——宁可少一行，也不让"没有 SKU 主键的行"[39m
[31m+  *    用 spuId/skuCode/条码/名称凑身份（那样会重新长出重复 :key 与写错对象）；[39m
[31m+  * 4. 同 skuId 重复出现只保留首行（分页边界/后端 JOIN 放大时的防御），同样写守卫日志。[39m
[31m+  */[39m
[31m+ export function mapProductRecords(records: any[]): ProductInfo[] {[39m
[31m+   const out: ProductInfo[] = [][39m
[31m+   const seen = new Set<number>()[39m
[31m+   for (const r of records) {[39m
[31m+     const skuId = toNumericId(r.skuId)[39m
[31m+     const spuId = toNumericId(r.spuId)[39m
[31m+     if (!skuId) {[39m
[31m+       guardNote([39m
[31m+         `[SKU维度守卫] 列表行缺正整数 skuId，已跳过（不得用 spuId/skuCode/barcode/name 顶替 SKU 主键）：spuId=${spuId || '-'} skuCode=${r?.skuCode ?? '-'} name=${r?.name ?? '-'}`[39m
[31m+       )[39m
[31m+       continue[39m
[31m+     }[39m
[31m+     if (seen.has(skuId)) {[39m
[31m+       guardNote(`[SKU维度守卫] 列表内 skuId=${skuId} 重复出现，仅保留首行（不跨 SKU 合并库存/价格）`)[39m
[31m+       continue[39m
[31m+     }[39m
[31m+     seen.add(skuId)[39m
[31m+     out.push({[39m
[31m+       id: skuId,[39m
[31m+       spuId,[39m
[31m+       skuId,[39m
[31m+       name: r.name ?? '',[39m
[31m+       categoryId: r.categoryId,[39m
[31m+       categoryName: r.categoryName,[39m
[31m+       brandName: r.brandName ?? '',[39m
[31m+       barcode: r.barcode ?? '',[39m
[31m+       price: Number(r.retailPrice ?? r.price ?? 0),[39m
[31m+       wholesalePrice: r.wholesalePrice != null ? Number(r.wholesalePrice) : Number(r.price ?? 0),[39m
[31m+       retailPrice: r.retailPrice != null ? Number(r.retailPrice) : Number(r.price ?? 0),[39m
[31m+       // 门店价 / 小程序价：与后端五档价格体系一致（列表接口已返回）[39m
[31m+       storePrice: r.storePrice != null ? Number(r.storePrice) : undefined,[39m
[31m+       miniappPrice: r.miniappPrice != null ? Number(r.miniappPrice) : undefined,[39m
[31m+       // 成本价：权限剥离后为 null，由页面如实展示（不能当作 0）[39m
[31m+       costPrice: r.costPrice != null ? Number(r.costPrice) : null,[39m
[31m+       // 库存取该 SKU 行自己的 available_qty（t_inventory_balance 按 ib.sku_id JOIN），不得累加别的规格[39m
[31m+       stock: Number(r.availableQty ?? r.stock ?? 0),[39m
[31m+       unit: r.unit ?? '',[39m
[31m+       image: r.mainImage ?? r.image,[39m
[31m+       specs: r.specs,[39m
[31m+       origin: r.origin,[39m
[31m+       safetyStock: r.warningThreshold != null ? Number(r.warningThreshold) : undefined,[39m
[31m+       // 后端状态枚举为大写 ON_SALE / OFF_SALE / DRAFT，上架 → ON，其余 → OFF[39m
[31m+       status: r.status === 'ON_SALE' ? 'ON' : 'OFF',[39m

[31m+       allowOnlineSale: r.allowOnlineSale,[39m
[31m+     })[39m
[31m+   }[39m
[31m+   return out[39m
[31m+ }[39m
